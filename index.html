-- ① Products table create table if not exists products ( id text primary key, name text not null, code text, price integer not null default 0, category text, images text[], sizes text[] default array['S','M','L','XL'], description text, created_at timestamptz default now() ); alter table products add column if not exists section text default 'new_arrivals'; alter table products add column if not exists stock integer default 0; alter table products add column if not exists cost_price integer default 0; alter table products add column if not exists discount_price integer default 0; alter table products disable row level security; -- ② Categories table create table if not exists categories ( id bigint generated always as identity primary key, name text not null, image text, created_at timestamptz default now() ); alter table categories disable row level security; -- ③ Orders table create table if not exists orders ( id bigint generated always as identity primary key, product_name text, product_id text, product_image text, size text, quantity integer default 1, price integer default 0, total integer default 0, delivery_charge integer default 80, shipping_zone text default 'inside', payment_type text default 'full', payment_method text default 'bkash', amount_paid integer default 0, sender_number text, transaction_id text, customer_name text, customer_phone text, customer_address text, note text, status text default 'new', created_at timestamptz default now() ); alter table orders disable row level security;
